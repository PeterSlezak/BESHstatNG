    'planning/BESHStatNG_CrossPlatform_MasterPlan.md','planning/BESHStatNG_CrossPlatform_MasterPlan2.12.md','planning/BESHStatNG_CrossPlatform_MasterPlan2.14.md','AGENTS.md'])

for name in sorted(selected):
    reader.safe(name);path=ROOT/name;data=path.read_bytes();raw=pin(path)
    reasons=[]
    try:findings=policy.privacy(name,data)
    except ValueError as error:
        assert 'Unattested privacy finding' in str(error);reasons.append('NARROW_POLICY_FINDING');findings=[]

    except UnicodeDecodeError:broad=[dict(category='NON_UTF8_RAW_SOURCE;DERIVATIVE_NOT_ORIGINAL_BYTES')]
    if broad:reasons.append('ACTUAL13_PAYLOAD_FINDINGS')
    if reasons:
        try:review=(json.dumps(privacy2['redact'](clean(json.loads(data))),indent=2,ensure_ascii=False)+'\n').encode()
        except (ValueError,UnicodeError):review=privacy2['redact_string'](clean(data.decode('utf-8-sig',errors='replace'))).encode()
        derivative=HERE/'publication/derivatives'/(name+'.review.json' if name.endswith('.json') else name+'.review.txt')
        try:
            findings=policy.privacy(derivative.relative_to(ROOT).as_posix(),review)
            assert not privacy2['scan'](derivative.relative_to(ROOT).as_posix(),review)
        except (ValueError,AssertionError):
            review=(json.dumps(dict(raw=raw,transport_only='Sensitive or non-UTF8 raw values omitted from derivative transport; local original retained. Not executable raw evidence.'),indent=2)+'\n').encode()

        derivative.parent.mkdir(parents=True,exist_ok=True)
        with derivative.open('xb') as stream:stream.write(review)
        exclusions.append(raw);derivatives.append(dict(raw=raw,derivative=pin(derivative),reasons=reasons,
            broad_raw_findings=len(broad),raw_original_retained=True,waiver=False))
        name=derivative.relative_to(ROOT).as_posix();path=derivative;data=review
    assert not privacy2['scan'](name,data)
    rows.append(dict(pin(path),crc32=format(zlib.crc32(data)&0xffffffff,'08x')))
    decisions.append(dict(path=name,sha256=rows[-1]['sha256'],findings=findings))
