        Public Shared Function BuildRocPointsTable(cutoffTable As Object(,), isLowerDirection As Boolean, tryFinite As UdfModelProfilePreparation38.FiniteDoubleReader) As Object(,)
            If cutoffTable Is Nothing OrElse cutoffTable.GetLength(0) < 2 Then
                Return UdfOutputTablesReports38.BuildSimpleNoteTable("ROC", "No ROC cut-off table was returned.").ToOutputModel().Values
            End If

            Dim n As Integer = cutoffTable.GetLength(0) - 1
            Dim out(n + 2, 4) As Object
            out(0, 0) = "Threshold"
            out(0, 1) = "Sensitivity"
            out(0, 2) = "Specificity"
            out(0, 3) = "FalsePositiveRate"
            out(0, 4) = "TruePositiveRate"

            out(1, 0) = ""
            out(1, 1) = 1.0R
            out(1, 2) = 0.0R
            out(1, 3) = 1.0R
            out(1, 4) = 1.0R

            For i As Integer = 1 To n
                Dim thresholdObj As Object = cutoffTable(i, 0)
                Dim threshold As Double
                If tryFinite(thresholdObj, threshold) Then
                    out(i + 1, 0) = If(isLowerDirection, -threshold, threshold)
                Else
                    out(i + 1, 0) = thresholdObj
                End If

                Dim sens As Double = Convert.ToDouble(cutoffTable(i, 1), CultureInfo.InvariantCulture)
                Dim spec As Double = Convert.ToDouble(cutoffTable(i, 2), CultureInfo.InvariantCulture)
                out(i + 1, 1) = sens
                out(i + 1, 2) = spec
                out(i + 1, 3) = 1.0R - spec
                out(i + 1, 4) = sens
            Next

            out(n + 2, 0) = ""
            out(n + 2, 1) = 0.0R
            out(n + 2, 2) = 1.0R
            out(n + 2, 3) = 0.0R
            out(n + 2, 4) = 0.0R
            Return out
        End Function

        Public Shared Function BuildRocStatsTable(summaryTable As Object(,)) As Object(,)

            Next
            Return out
        End Function

        Public Shared Function BuildRosnerTable(alpha As Double, outliers() As Double, addCaution As Boolean) As UdfReportProjection38
            Dim nOut As Integer = If(outliers Is Nothing, 0, outliers.Length)
            Dim rows As Integer = 3 + Math.Max(1, nOut)
            Dim body(rows - 1, 1) As Object
            body(0, 0) = "Alpha"
            body(0, 1) = alpha
            body(1, 0) = "Number of outliers"
            body(1, 1) = nOut
            body(2, 0) = "List of outliers"
            If nOut > 0 Then
                body(2, 1) = outliers(0)
            Else
                body(2, 1) = ""
            End If
            For i As Integer = 1 To nOut - 1
                body(2 + i, 0) = ""
                body(2 + i, 1) = outliers(i)
            Next
            If nOut = 0 Then
                body(3, 0) = ""
                body(3, 1) = ""
            End If

            Dim t As New ResultTable
            t.SetBody(body)
            t.AddHeaderTopRow({"Rosner Generalized ESD Test", ""})
            If addCaution Then
                t.AddFootnote("Interpret with caution for sample sizes below 25.")
            End If
            Return New UdfReportProjection38(t.ToOutputModel(), UdfReportPreparation38.NormalizeBlanks)
        End Function

        Public Shared Function ResolveAvailableAxisCount(preferredAxisCount As Integer, valueSelector As Func(Of Integer, Double())) As Integer
            If preferredAxisCount < 1 OrElse valueSelector Is Nothing Then Return 0
            Dim available As Integer = 0
            For axisIdx As Integer = 0 To preferredAxisCount - 1
                Try
                    Dim values() As Double = valueSelector(axisIdx)
                    If values Is Nothing Then Exit For
                    available += 1
                Catch ex As Exception
                    Exit For
                End Try
            Next
            Return available
        End Function

        Public Shared Function BuildLabeledMatrixOutput(matrix(,) As Double,
                                                  rowNames() As String,
                                                  colNames() As String,
                                                  includeHeader As Boolean) As UdfReportProjection38

            If matrix Is Nothing Then Return UdfReportProjection38.FromFault(UdfReportFault38.NotAvailable)

            Dim nRows As Integer = matrix.GetLength(0)
            Dim nCols As Integer = matrix.GetLength(1)

            If rowNames Is Nothing OrElse rowNames.Length <> nRows Then
                ReDim rowNames(nRows - 1)
                For i As Integer = 0 To nRows - 1
                    rowNames(i) = "Row" & (i + 1).ToString(CultureInfo.InvariantCulture)
                Next
            End If

            If colNames Is Nothing OrElse colNames.Length <> nCols Then
                ReDim colNames(nCols - 1)
                For j As Integer = 0 To nCols - 1
                    colNames(j) = "Col" & (j + 1).ToString(CultureInfo.InvariantCulture)
                Next
            End If

            If Not includeHeader Then
                Dim outNoHdr(nRows - 1, nCols - 1) As Object
                For i As Integer = 0 To nRows - 1
                    For j As Integer = 0 To nCols - 1
                        outNoHdr(i, j) = UdfReportProjection38.SafeNumber(matrix(i, j))
                    Next
                Next
                Return New UdfReportProjection38(New Global.BESHStatNG.ResultTableOutputModel(outNoHdr), UdfReportPreparation38.Raw)
            End If

            Dim out(nRows, nCols) As Object
            out(0, 0) = ""

            For j As Integer = 0 To nCols - 1
                out(0, j + 1) = colNames(j)
            Next

            For i As Integer = 0 To nRows - 1
                out(i + 1, 0) = rowNames(i)
                For j As Integer = 0 To nCols - 1
                    out(i + 1, j + 1) = UdfReportProjection38.SafeNumber(matrix(i, j))
                Next
            Next

            Return New UdfReportProjection38(New Global.BESHStatNG.ResultTableOutputModel(out), UdfReportPreparation38.Raw)
        End Function

        Public Shared Function BuildNamedMatrixOutput(idHeader As String, rowNames() As String, colNames() As String,
                                               mat(,) As Double, includeHeader As Boolean) As UdfReportProjection38
            If mat Is Nothing Then Return UdfReportProjection38.FromFault(UdfReportFault38.NotAvailable)

            Dim n As Integer = mat.GetLength(0)
            Dim p As Integer = mat.GetLength(1)
            Dim out(n - 1 + If(includeHeader, 1, 0), p) As Object
            Dim r0 As Integer = 0

            If includeHeader Then
                out(0, 0) = idHeader
                For j As Integer = 0 To p - 1
                    out(0, j + 1) = If(colNames IsNot Nothing AndAlso j < colNames.Length, colNames(j), (j + 1).ToString(CultureInfo.InvariantCulture))
                Next
                r0 = 1
            End If

            For i As Integer = 0 To n - 1
                out(r0 + i, 0) = If(rowNames IsNot Nothing AndAlso i < rowNames.Length, rowNames(i), (i + 1).ToString(CultureInfo.InvariantCulture))
                For j As Integer = 0 To p - 1
                    out(r0 + i, j + 1) = mat(i, j)
                Next
            Next

            Return New UdfReportProjection38(New Global.BESHStatNG.ResultTableOutputModel(out), UdfReportPreparation38.NormalizeBlanks)
        End Function

        Public Shared Function BuildWelchAnovaTable(data()() As Double, names() As String) As Global.BESHStatNG.ResultTable
            Dim mdl As New Global.BESHStatNG.parametric.OneWayANOVA(data, names)
            mdl.compute()

            Dim body(0, 3) As Object
            body(0, 0) = data.Length - 1
            body(0, 1) = w.DF1
            body(0, 2) = w.TestStatistics1

            t.AddHeaderTopRow({"Source", "df numerator", "df denominator", "F", "P-value"})
            Return t
        End Function

        ' Invalid selectors are resolved only AFTER construction/compute, as originally.
        ' wrapResults always supplies a non-null List in the pinned Core engine.
        Public Shared Function BuildOneWayMcpTables(data()() As Double, names() As String, which As String, alphaValue As Double, ByRef stackAll As Boolean) As List(Of Global.BESHStatNG.ResultTable)
            Dim mdl As New Global.BESHStatNG.parametric.OneWayANOVA(data, names)
            mdl.compute()
            stackAll = False
            Select Case LegacyUdfText38.CaseToken(which, New String() {"", "TUKEY", "TUKEYKRAMER", "TUKEY-KRAMER", "TK", "GAMESHOWELL", "GAMES-HOWELL", "GH", "GAMES", "LSD", "FISHER", "FISHERLSD", "FISHER-LSD", "BONF", "BONFERRONI", "ALL", "BOTH"})
                Case "", "TUKEY", "TUKEYKRAMER", "TUKEY-KRAMER", "TK"
                    mdl.TukeyKramer(alphaValue)
                Case "GAMESHOWELL", "GAMES-HOWELL", "GH", "GAMES"
                    mdl.GamesHowell(alphaValue)
                Case "LSD", "FISHER", "FISHERLSD", "FISHER-LSD"
                    mdl.FisherLSD(False, alphaValue)
                Case "BONF", "BONFERRONI"
                    mdl.FisherLSD(True, alphaValue)
                Case "ALL", "BOTH"
                    mdl.FisherLSD(False, alphaValue)
                    mdl.FisherLSD(True, alphaValue)
                    mdl.TukeyKramer(alphaValue)
                    mdl.GamesHowell(alphaValue)
                    stackAll = True
                Case Else
                    Return Nothing
            End Select
            Return mdl.wrapResults()
        End Function
