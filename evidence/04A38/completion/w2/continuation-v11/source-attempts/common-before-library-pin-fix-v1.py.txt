def sha(data):
    return hashlib.sha256(data).hexdigest()

def pin(path):
    path = Path(path)
    with path.open('rb') as stream:
        digest = hashlib.file_digest(stream, 'sha256').hexdigest()
    return dict(path=path.relative_to(ROOT).as_posix(), bytes=path.stat().st_size, sha256=digest)

def library_pin(path):
    result = pin(path)
    return dict(name=Path(path).name, bytes=result['bytes'], sha256=result['sha256'])

def save(name, value):
    target = HERE / name
    target.parent.mkdir(parents=True, exist_ok=True)
    with target.open('x', encoding='utf-8', newline='\n') as stream:
        json.dump(value, stream, ensure_ascii=False, indent=2, allow_nan=False)
        stream.write('\n')

def git(*args):

def compiler(name):
    locator = Path(os.environ['ProgramFiles(x86)']) / 'Microsoft Visual Studio/Installer/vswhere.exe'
    return subprocess.check_output([str(locator), '-products', '*', '-version', '[18.0,19.0)',

def write_source(path, text):
    target = ROOT / path
    before = target.read_bytes()
    newline = '\r\n' if b'\r\n' in before else '\n'
    target.write_bytes((b'\xef\xbb\xbf' if before.startswith(b'\xef\xbb\xbf') else b'') +
        text.replace('\r\n', '\n').replace('\n', newline).encode('utf-8'))
