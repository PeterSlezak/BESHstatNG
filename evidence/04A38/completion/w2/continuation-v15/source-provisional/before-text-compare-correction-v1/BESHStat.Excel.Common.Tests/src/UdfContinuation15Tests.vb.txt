    <DataTestMethod, DataRow(False), DataRow(True)>
    Public Sub CoxRecordsPreserveInputOrderIdentityTiesAndCensorValues(stratified As Boolean)
        Dim data As New CoxPHData With {
            .TimeData = New Double() {5, 2, 5}, .CensorData = New Integer() {1, 0, 2},
            .RowIds = New Integer() {42, 8, 42}, .bStrata = stratified,
            .StrataData = New String() {"B", Nothing, "A"}}
        Dim x As Double(,) = {{1, 2}, {3, 4}, {5, Double.NaN}}
        Dim rows = UdfCoxInputPreparation38.PrepareRecords(data, 3, 2, x)
        Assert.AreEqual(3, rows.Count)
        Assert.AreEqual(42, rows(0).Index)
        Assert.AreEqual(8, rows(1).Index)
        Assert.AreEqual(42, rows(2).Index)
        Assert.AreEqual(2, rows(2).Censorship)
        Assert.AreEqual(5.0R, rows(2).Time)
        Assert.IsTrue(Double.IsNaN(rows(2).Covariates(1)))
        Assert.AreEqual(If(stratified, "B", "__ALL__"), rows(0).Stratum)
        Assert.AreEqual(If(stratified, Nothing, "__ALL__"), rows(1).strStratum)
        Assert.AreNotSame(rows(0).Covariates, rows(1).Covariates)
        x(0, 0) = 99
        Assert.AreEqual(1.0R, rows(0).Covariates(0))
    End Sub

    <DataTestMethod, DataRow(False), DataRow(True)>
    Public Sub CoxSummaryPreservesRobustSelectionAndDisplayCallbackOrder(robust As Boolean)
        Dim result As New CoxResult With {.Coefficients = New Double() {1}, .VarCov = New Double(,) {{1}}, .VarCovRobust = New Double(,) {{4}}}
        Dim calls As New List(Of Double)
        Dim table = UdfCoxGeeReportPreparation38.BuildCoxSummaryTable(True, 1, New String() {"Dose"}, result, robust, 2, "95,5%",
            Function(value)
                calls.Add(value)
                Return CType(value, Object)
            End Function)
        Assert.AreEqual(2, table.GetLength(0))
        Assert.AreEqual(8, table.GetLength(1))
        Assert.AreEqual("HR 95,5% LCL", table(0, 6))
        Assert.AreEqual(If(robust, 2.0R, 1.0R), table(1, 2))
        CollectionAssert.AreEqual(New Double() {1, If(robust, -3.0R, -1.0R), If(robust, 5.0R, 3.0R)}, calls.ToArray())
    End Sub

    Private Class ThrowingKey
        Public Calls As Integer
        Public Overrides Function ToString() As String
            Calls += 1
            If Calls = 2 Then Throw New InvalidOperationException("second key conversion")
            Return "first"
        End Function
    End Class

    <TestMethod>
    Public Sub CoxBaselinePreservesRepeatedKeyConversionAndPartialByRefRow()
        Dim key As New ThrowingKey
        Dim table = UdfCoxGeeReportPreparation38.CreateCoxBaselineTable(3, "table")
        Dim row As Integer = 1
        Assert.ThrowsException(Of InvalidOperationException)(
            Sub() UdfCoxGeeReportPreparation38.AppendCoxBaselineRows(table, New Double(,) {{1, 0.9, 0.1}, {2, 0.8, 0.2}}, key, "table", row))
        Assert.AreEqual(2, key.Calls)
        Assert.AreEqual(2, row)
        Assert.AreEqual("first", table(1, 0))
        Assert.IsNull(table(2, 0))
    End Sub

    <TestMethod>
    Public Sub GeeProfileDefaultsAndListAliasingAreDistinctPerOwnerAndInstance()
        Dim first As New UdfGeeAtProfile38(Of UdfLsmProfileValue38)
        Dim second As New UdfGeeAtProfile38(Of UdfLsmProfileValue38)
        Dim component As New UdfGeeComponent38(Of UdfLsmProfileValue38)
        Assert.IsFalse(first.TimeSpecified)
        Assert.IsTrue(Double.IsNaN(first.TimeValue))

        Assert.AreNotSame(first.ProfileValues, component.ProfileValues)

        first.ProfileValues = aliasList
        Assert.AreSame(aliasList, first.ProfileValues)
        first.TimeSpecified = True
        first.TimeValue = 3
        Assert.IsTrue(first.TimeSpecified)
        Assert.AreEqual(3.0R, first.TimeValue)
    End Sub
