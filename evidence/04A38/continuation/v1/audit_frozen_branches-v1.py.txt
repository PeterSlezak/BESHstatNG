def children(encoded):
 """Split only a recorded outer array, retaining every scalar token unchanged."""
 match=re.fullmatch(r'([^\[{}]+)\[([0-9:,\-]+)\]\{(.*)\}',encoded,re.S)
 if not match:return None
 bounds=[tuple(map(int,pair.split(':'))) for pair in match[2].split(',')]
 result=[];depth=0;begin=0;body=match[3]
 for index,char in enumerate(body):
  if char=='{':depth+=1
  elif char=='}':depth-=1
  elif char=='|' and depth==0:result.append(body[begin:index]);begin=index+1
 if body:result.append(body[begin:])
 assert depth==0
 cardinality=1
 for _,length in bounds:cardinality*=length
 assert cardinality==len(result)
 return dict(element_type=match[1],bounds=bounds,values=result)
