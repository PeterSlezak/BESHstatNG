"""Execute only original references, never a current supplemental comparison."""
import json,os,subprocess,sys
from pathlib import Path
from prepare_original import ROOT,HERE,PRIVATE,sha,pin,save
sys.path.insert(0,str(ROOT/'evidence/04A38'))
from run import m,PRIVATE as RUNROOT
from baseline_report import clean

def main():
 config,mode=sys.argv[1:3];assert config in ['Debug','Release'] and mode in ['plain','repeat','trace']
 seal=json.loads((HERE/'input-seal.json').read_bytes())
 for row in seal['files']:assert pin(ROOT/row['path'])==row
 source=PRIVATE/('original-'+config+('-trace' if mode=='trace' else '-plain'))
 out=HERE/'original-results'/config/mode;out.mkdir(parents=True,exist_ok=False)
 locator=Path(os.environ['ProgramFiles(x86)'])/'Microsoft Visual Studio/Installer/vswhere.exe'
 compiler=subprocess.check_output([str(locator),'-products','*','-version','[18.0,19.0)','-find','MSBuild/Current/Bin/Roslyn/csc.exe'],text=True).splitlines()[0]
 facade=Path(os.environ['ProgramFiles(x86)'])/'Reference Assemblies/Microsoft/Framework/.NETFramework/v4.8/Facades/netstandard.dll'
 exe=source/('LateCapture38-'+mode+'.exe');runner=m.Runner(RUNROOT)
 result=runner.command('late38-original-'+config+'-'+mode+'-recorder',[compiler,'/nologo','/target:exe','/reference:System.Core.dll','/reference:System.Web.Extensions.dll','/reference:'+str(source/'BESHStat.Core.dll'),'/reference:'+str(facade),'/out:'+str(exe),str(HERE/'LateCapture38.cs')]);assert result['exit_code']==0
 result=runner.command('late38-original-'+config+'-'+mode+'-execute',[str(exe),str(source),str(HERE/('inputs-'+config+'.jsonl')),str(out/'results.tsv'),str(out/'execution.json'),'original'],env={'BESHSTAT_LATE38_CONFIGURATION':config});assert result['exit_code']==0
 save('original-execution-'+config+'-'+mode+'.json',dict(status='ORIGINAL_SOURCE_EXECUTED_NOT_YET_SEALED',configuration=config,mode=mode,recorder=pin(HERE/'LateCapture38.cs'),
  executable=dict(name=exe.name,bytes=exe.stat().st_size,sha256=sha(exe.read_bytes())),results=pin(out/'results.tsv'),execution=pin(out/'execution.json'),command=clean(result),
  current_implementation_executed=False,reference_class='LATE_ARCHIVED_ENTRY_SUPPLEMENT',chronology='CAPTURED_AFTER_INITIAL_04A38_IMPLEMENTATION_FROM_PINNED_ENTRY'))

if __name__=='__main__':main()
