class InvalidEvidence(ValueError):pass
def need(value,message):
 if not value:raise InvalidEvidence(message)
def digest(data):return hashlib.sha256(data).hexdigest()
def load(path):
 def unique(pairs):
  result={}
  for k,v in pairs:need(k not in result,'Duplicate JSON key');result[k]=v
  return result
 return json.loads(path.read_bytes(),object_pairs_hook=unique)
def check_pin(root,row):
 p=(root/row['path']).resolve();need(p.is_relative_to(root.resolve()),'Escaping source/input path')
 data=p.read_bytes();need(len(data)==row['bytes'] and digest(data)==row['sha256'],'Changed frozen field/source: '+row['path'])
